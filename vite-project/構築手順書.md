## ステップ2-3: シンプルな描画シェーダー

`src/shaders/simple.frag`:

```glsl
uniform vec2 uMouse;      // マウス位置（-1〜1）
uniform vec2 uResolution; // 画面サイズ

varying vec2 vUv;

void main() {
  // UV座標を-1〜1に変換
  vec2 coord = vUv * 2.0 - 1.0;

  // アスペクト比補正
  float aspect = uResolution.x / uResolution.y;
  coord.x *= aspect;

  vec2 mouse = uMouse;
  mouse.x *= aspect;

  // マウスからの距離を計算
  float dist = length(coord - mouse);

  // 距離に応じて色を決定（近いほど明るい）
  float brightness = smoothstep(0.2, 0.0, dist);
  vec3 color = vec3(brightness);

  gl_FragColor = vec4(color, 1.0);
}
```

**このシェーダーの役割:**
- マウス位置を受け取る
- 各ピクセルとマウスの距離を計算
- 距離が近いピクセルを明るく表示

## ステップ2-4: main.tsを更新

`src/main.ts` を以下に置き換え：

```typescript
import * as THREE from 'three';
import vertexShader from './shaders/simple.vert?raw';
import fragmentShader from './shaders/simple.frag?raw';

const canvas = document.getElementById('canvas') as HTMLCanvasElement;
const renderer = new THREE.WebGLRenderer({ canvas });
renderer.setSize(window.innerWidth, window.innerHeight);

const scene = new THREE.Scene();
const camera = new THREE.OrthographicCamera(-1, 1, 1, -1, 0, 1);

// シェーダーマテリアル
const material = new THREE.ShaderMaterial({
  uniforms: {
    uMouse: { value: new THREE.Vector2(0, 0) },
    uResolution: { value: new THREE.Vector2(window.innerWidth, window.innerHeight) },
  },
  vertexShader,
  fragmentShader,
});

// フルスクリーンの四角形
const geometry = new THREE.PlaneGeometry(2, 2);
const mesh = new THREE.Mesh(geometry, material);
scene.add(mesh);

// マウス座標を-1〜1に変換
function updateMouse(clientX: number, clientY: number) {
  const x = (clientX / window.innerWidth) * 2 - 1;
  const y = -((clientY / window.innerHeight) * 2 - 1);
  material.uniforms.uMouse.value.set(x, y);
}

// マウスイベント
canvas.addEventListener('mousemove', (e) => {
  updateMouse(e.clientX, e.clientY);
});

// アニメーションループ
function animate() {
  requestAnimationFrame(animate);
  renderer.render(scene, camera);
}

animate();
```

# フェーズ3: フィードバック効果

ここが重要です。描いた線を残像として残す「ピンポンバッファ」を実装します。

## ステップ3-1: ピンポンバッファの概念理解

**ピンポンバッファとは？**

```
フレームN:   rtA（前の絵） → シェーダー → rtB（新しい絵）
フレームN+1: rtB（前の絵） → シェーダー → rtA（新しい絵）
```

2つのレンダーターゲット（rtA, rtB）を交互に使うことで、
前フレームの結果を次フレームの入力にできます。

## ステップ3-2: フィードバックシェーダー

`src/shaders/feedback.frag` を作成：

```glsl
uniform sampler2D tPrev;  // 前フレームの絵
uniform vec2 uMouse;
uniform vec2 uPrevMouse;  // 前フレームのマウス位置
uniform vec2 uResolution;
uniform float uDecay;     // 減衰率（0.0〜1.0）

varying vec2 vUv;

// 点から線分までの距離を計算
float lineDist(vec2 p, vec2 a, vec2 b) {
  vec2 pa = p - a;
  vec2 ba = b - a;
  float h = clamp(dot(pa, ba) / (dot(ba, ba) + 0.0001), 0.0, 1.0);
  return length(pa - ba * h);
}

void main() {
  // 前フレームの色を取得して減衰させる
  vec3 prevColor = texture2D(tPrev, vUv).rgb;
  vec3 faded = prevColor * uDecay;

  // 座標変換とアスペクト比補正
  vec2 coord = vUv * 2.0 - 1.0;
  float aspect = uResolution.x / uResolution.y;
  coord.x *= aspect;

  vec2 mouse = uMouse;
  mouse.x *= aspect;
  vec2 prevMouse = uPrevMouse;
  prevMouse.x *= aspect;

  // マウスの軌跡（線分）までの距離
  float dist = lineDist(coord, mouse, prevMouse);

  // 線を描画（距離が近いほど明るい）
  float line = smoothstep(0.02, 0.0, dist);
  vec3 lineColor = vec3(line);

  // 前フレームの絵 + 新しい線
  vec3 finalColor = faded + lineColor;

  gl_FragColor = vec4(finalColor, 1.0);
}
```

**重要なポイント:**
- `tPrev`: 前フレームの絵を受け取る
- `uDecay`: 0.95なら前フレームの95%の明るさで残る
- `lineDist`: マウスの移動軌跡（線分）を描画

## ステップ3-3: main.tsを更新（ピンポンバッファ実装）

`src/main.ts` を以下に置き換え：

```typescript
import * as THREE from 'three';
import vertexShader from './shaders/simple.vert?raw';
import fragmentShader from './shaders/feedback.frag?raw';

const canvas = document.getElementById('canvas') as HTMLCanvasElement;
const renderer = new THREE.WebGLRenderer({ canvas });
renderer.setSize(window.innerWidth, window.innerHeight);

const scene = new THREE.Scene();
const camera = new THREE.OrthographicCamera(-1, 1, 1, -1, 0, 1);

// レンダーターゲット（ピンポンバッファ）
function createRenderTarget() {
  return new THREE.WebGLRenderTarget(window.innerWidth, window.innerHeight, {
    minFilter: THREE.LinearFilter,
    magFilter: THREE.LinearFilter,
  });
}

let rtA = createRenderTarget();
let rtB = createRenderTarget();

// シェーダーマテリアル
const material = new THREE.ShaderMaterial({
  uniforms: {
    tPrev: { value: null },
    uMouse: { value: new THREE.Vector2(-2, -2) },
    uPrevMouse: { value: new THREE.Vector2(-2, -2) },
    uResolution: { value: new THREE.Vector2(window.innerWidth, window.innerHeight) },
    uDecay: { value: 0.95 },  // 95%残る = ゆっくり消える
  },
  vertexShader,
  fragmentShader,
});

const geometry = new THREE.PlaneGeometry(2, 2);
const mesh = new THREE.Mesh(geometry, material);
scene.add(mesh);

// マウス管理
const mouse = { x: -2, y: -2, prevX: -2, prevY: -2 };

function updateMouse(clientX: number, clientY: number) {
  mouse.prevX = mouse.x;
  mouse.prevY = mouse.y;
  mouse.x = (clientX / window.innerWidth) * 2 - 1;
  mouse.y = -((clientY / window.innerHeight) * 2 - 1);
}

canvas.addEventListener('mousemove', (e) => {
  updateMouse(e.clientX, e.clientY);
});

// アニメーションループ
function animate() {
  requestAnimationFrame(animate);

  // Uniformを更新
  material.uniforms.uMouse.value.set(mouse.x, mouse.y);
  material.uniforms.uPrevMouse.value.set(mouse.prevX, mouse.prevY);

  // === ピンポンバッファの処理 ===
  // 1. rtAを読み込んで、rtBに描画
  material.uniforms.tPrev.value = rtA.texture;
  renderer.setRenderTarget(rtB);
  renderer.render(scene, camera);

  // 2. rtBの結果を画面に表示
  material.uniforms.tPrev.value = rtB.texture;
  renderer.setRenderTarget(null);
  renderer.render(scene, camera);

  // 3. rtAとrtBを入れ替え
  [rtA, rtB] = [rtB, rtA];
}

animate();
```

## ステップ4-1: 色相シフトの追加

`src/shaders/feedback.frag` に色変換関数を追加：

```glsl
// ファイルの先頭に追加

// RGB → HSV変換
vec3 rgb2hsv(vec3 c) {
  vec4 K = vec4(0.0, -1.0 / 3.0, 2.0 / 3.0, -1.0);
  vec4 p = mix(vec4(c.bg, K.wz), vec4(c.gb, K.xy), step(c.b, c.g));
  vec4 q = mix(vec4(p.xyw, c.r), vec4(c.r, p.yzx), step(p.x, c.r));
  float d = q.x - min(q.w, q.y);
  return vec3(abs(q.z + (q.w - q.y) / (6.0 * d + 1.0e-10)), d / (q.x + 1.0e-10), q.x);
}

// HSV → RGB変換
vec3 hsv2rgb(vec3 c) {
  vec4 K = vec4(1.0, 2.0 / 3.0, 1.0 / 3.0, 3.0);
  vec3 p = abs(fract(c.xxx + K.xyz) * 6.0 - K.www);
  return c.z * mix(K.xxx, clamp(p - K.xxx, 0.0, 1.0), c.y);
}
```

uniforms に追加：

```glsl
uniform float uHueShift;  // 色相シフト速度
uniform float uTime;      // 経過時間
```

main関数を更新：

```glsl
void main() {
  // 前フレームの色を取得
  vec3 prevColor = texture2D(tPrev, vUv).rgb;

  // HSVに変換して色相をシフト
  vec3 hsv = rgb2hsv(prevColor);
  hsv.x = fract(hsv.x + uHueShift * 0.001);  // 色相を少しずつ変化
  vec3 faded = hsv2rgb(hsv) * uDecay;

  // ... 以下は同じ（マウス描画部分）

  // 新しく描く線にも虹色を適用
  float hue = fract(uTime * 0.1);
  vec3 neonColor = hsv2rgb(vec3(hue, 1.0, 1.0));
  vec3 lineColor = neonColor * line;

  vec3 finalColor = faded + lineColor;
  gl_FragColor = vec4(finalColor, 1.0);
}
```

main.tsにuniformを追加：

```typescript

}
```

## ステップ4-2: グロー効果の追加

feedback.fragのline計算部分を更新：

```glsl
// 距離に基づくグロー
float line = smoothstep(0.015, 0.005, dist);
float glow = 0.01 / (dist + 0.01);
glow = pow(glow, 2.0) * 0.3;

```

**グローの仕組み:**
- `1/距離` で距離が近いほど明るくなる
- `pow(glow, 2.0)` で光の減衰をリアルに

## ステップ4-3: ブルーム効果（オプション）

よりリッチな見た目にしたい場合、ポストプロセッシング用の別シェーダーを追加できます。

`src/shaders/output.frag`:

```glsl
uniform sampler2D tDiffuse;
uniform vec2 uResolution;
varying vec2 vUv;

void main() {
  vec3 color = texture2D(tDiffuse, vUv).rgb;

  // 簡易ブルーム（周囲をサンプリング）
  vec3 bloom = vec3(0.0);
  float blurSize = 3.0 / uResolution.x;

  for (int i = 0; i < 8; i++) {
    float angle = float(i) * 0.785; // 45度ずつ
    vec2 offset = vec2(cos(angle), sin(angle)) * blurSize;
    bloom += texture2D(tDiffuse, vUv + offset).rgb;
  }

  color += bloom * 0.1;

  // ガンマ補正
  color = pow(color, vec3(1.0 / 2.2));

  gl_FragColor = vec4(color, 1.0);
}
```

main.tsで2パスレンダリング：

```typescript
// output用マテリアルを追加
const outputMaterial = new THREE.ShaderMaterial({
  uniforms: {
    tDiffuse: { value: null },
    uResolution: { value: new THREE.Vector2(window.innerWidth, window.innerHeight) },
  },
  vertexShader,
  fragmentShader: outputFragShader,  // output.fragをインポート
});

function animate() {
  // パス1: フィードバック処理
  mesh.material = material;
  material.uniforms.tPrev.value = rtA.texture;
  renderer.setRenderTarget(rtB);
  renderer.render(scene, camera);

  // パス2: ブルーム適用して画面出力
  mesh.material = outputMaterial;
  outputMaterial.uniforms.tDiffuse.value = rtB.texture;
  renderer.setRenderTarget(null);
  renderer.render(scene, camera);

  [rtA, rtB] = [rtB, rtA];
}
```
